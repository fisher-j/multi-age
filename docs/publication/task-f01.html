F01 — fuel-component separation and model exploration

Status: Completed — analytical record. Working target: Reproducible component/model record.

This page owns the fuel-component derivation, exploration, candidate fitting, and provisional model-review record. The analytical sequence continues with RD04 inference and display and RT03 Results writing; those steps own their subsequent decisions.

Original objective and guardrails

Create publication-only fuel components that preserve the thesis analyses while making the pre- and post-PCT data usable as distinct reference phases. Retain the fixed experimental-design random-intercept structure; record every model candidate and diagnostic; do not treat an AIC leader as a selected model.

  • Thesis notebooks, raw data, and thesis-era .rda objects remain unchanged.
  • Pre-PCT litter and duff are separated using the Finney–Martin model-3 forest-floor partition; post-PCT litter and duff are deliberately omitted because their boundary is not currently defensible.
  • Woody and herbaceous vegetation are separate in both phases.
  • The primary PCT contrast is matched-transect pre-PCT minus post-PCT woody standing vegetation—an inferred potential surface-fuel contribution, not a direct slash-load measurement.

01 — Derive components and audit pairing

Purpose. Separate reportable fuel components and construct matched transect records.

Rationale. Partition pre-PCT forest floor with Finney–Martin model 3, bound negative shallow litter extrapolations at zero, keep woody/herbaceous cover-height loads separate, and carry exported deadwood loads forward unchanged. Omit post-PCT litter/duff; standing woody reduction is potential contribution, not measured slash.

Calculation — f01-01-derive-components 
source("scripts/fuel/00-fuel-config.R")
local({
# Finney and Martin (1993), model 3: cumulative forest-floor mass (Mg ha^-1)
# from the surface to depth ds_cm within a profile of total depth dp_cm.
# The model is used only for the pre-PCT forest-floor measurements, where the
# recorded total depth and estimated litter proportion describe the forest floor.
# Its fitted range does not support a negative cumulative mass; the derivation
# script records and bounds such rare shallow-layer extrapolations at zero.
finney_martin_cumulative_load <- function(ds_cm, dp_cm) {
  result <- 6.461 * ds_cm^1.07 - 0.254 * dp_cm
  result[ds_cm == 0] <- 0
  result
}

publication_root <- normalizePath(getwd(), winslash = "/")
script_dir <- file.path(publication_root, "scripts", "fuel")
paths_common <- publication_paths(publication_root)

library(dplyr)
library(readr)
library(tidyr)

paths <- publication_fuel_paths(publication_root)
dir.create(paths$data, recursive = TRUE, showWarnings = FALSE)
dir.create(paths$manifests, recursive = TRUE, showWarnings = FALSE)

# Typed, checksum-validated exports from the separate thesis operation.
transects <- readRDS(paths$input)
legacy_total_load <- readRDS(paths$legacy_load_input)
ids <- publication_fuel_ids

# Keep the field measurements intact for audit. Post-PCT forest-floor values
# are retained here but deliberately receive no duff/litter load conversion.
forest_floor_station <- transects |>
  select(phase, all_of(ids), matches("^(duff_litter|pct_litter|fbd)[12]$")) |>
  pivot_longer(
    cols = matches("^(duff_litter|pct_litter|fbd)[12]$"),
    names_to = c(".value", "station"),
    names_pattern = "(.*?)([12])$"
  ) |>
  transmute(
    phase,
    across(all_of(ids)),
    station = as.integer(station),
    total_forest_floor_depth_cm = duff_litter,
    estimated_litter_fraction = pct_litter / 100,
    estimated_litter_depth_cm = total_forest_floor_depth_cm * estimated_litter_fraction,
    estimated_duff_depth_cm = total_forest_floor_depth_cm - estimated_litter_depth_cm,
    fuel_bed_depth_cm = fbd,
    forest_floor_load_eligible = phase == "prepct"
  )

if (any(forest_floor_station$estimated_litter_fraction < 0 |
  forest_floor_station$estimated_litter_fraction > 1, na.rm = TRUE)) {
  stop("Estimated litter fractions fall outside [0, 1].")
}

# Finney and Martin's cumulative curve partitions the pre-PCT total profile at
# the estimated litter/duff boundary. These are model-derived loads, not direct
# independent bulk-density measurements of the two layers.
forest_floor_pre_station <- forest_floor_station |>
  filter(forest_floor_load_eligible) |>
  mutate(
    total_forest_floor_load_mgha = finney_martin_cumulative_load(
      total_forest_floor_depth_cm,
      total_forest_floor_depth_cm
    ),
    litter_load_mgha = finney_martin_cumulative_load(
      estimated_litter_depth_cm,
      total_forest_floor_depth_cm
    ),
    litter_load_raw_mgha = litter_load_mgha,
    litter_load_bounded_at_zero = litter_load_raw_mgha < 0,
    litter_load_mgha = pmax(litter_load_raw_mgha, 0),
    duff_load_mgha = total_forest_floor_load_mgha - litter_load_mgha
  )

if (any(forest_floor_pre_station$duff_load_mgha < 0, na.rm = TRUE)) {
  stop("Finney-Martin component calculation produced a negative duff load; inspect the input depths before proceeding.")
}

forest_floor_components <- forest_floor_pre_station |>
  select(phase, all_of(ids), station, litter_load_mgha, duff_load_mgha) |>
  pivot_longer(
    cols = ends_with("_load_mgha"),
    names_to = "component",
    values_to = "load_mgha"
  ) |>
  mutate(
    component = recode(
      component,
      litter_load_mgha = "litter",
      duff_load_mgha = "duff"
    ),
    derivation = "Finney-Martin 1993 model 3 cumulative-load partition"
  ) |>
  summarise(
    load_mgha = mean(load_mgha),
    n_stations = sum(!is.na(load_mgha)),
    derivation = first(derivation),
    .by = c(all_of(ids), "phase", "component")
  )

# FireMon-style vegetation calculations remain separate by physiognomic class.
vegetation_station <- transects |>
  select(
    phase,
    all_of(ids),
    matches("^(live_woody|dead_woody|avg_w_ht|live_herb|dead_herb|avg_h_ht)[12]$")
  ) |>
  pivot_longer(
    cols = -c(phase, all_of(ids)),
    names_to = c(".value", "station"),
    names_pattern = "(.*?)([12])$"
  ) |>
  mutate(
    woody_height_m = if_else(live_woody == 0 & dead_woody == 0, 0, avg_w_ht),
    herb_height_m = if_else(live_herb == 0 & dead_herb == 0, 0, avg_h_ht),
    woody_load_mgha = ((live_woody + dead_woody) / 100) * woody_height_m * 18,
    herb_load_mgha = ((live_herb + dead_herb) / 100) * herb_height_m * 8
  ) |>
  select(phase, all_of(ids), station, woody_load_mgha, herb_load_mgha)

vegetation_components <- vegetation_station |>
  pivot_longer(
    cols = ends_with("_load_mgha"),
    names_to = "component",
    values_to = "load_mgha"
  ) |>
  mutate(
    component = recode(
      component,
      woody_load_mgha = "woody_vegetation",
      herb_load_mgha = "herbaceous_vegetation"
    ),
    derivation = "FireMon cover-height bulk-density calculation retained by component"
  ) |>
  summarise(
    load_mgha = mean(load_mgha),
    n_stations = sum(!is.na(load_mgha)),
    derivation = first(derivation),
    .by = c("phase", all_of(ids), "component")
  )

dead_woody_components <- legacy_total_load |>
  select(phase, all_of(ids), onehr, tenhr, hundhr, thoushr) |>
  pivot_longer(
    cols = c(onehr, tenhr, hundhr, thoushr),
    names_to = "component",
    values_to = "load_mgha"
  ) |>
  mutate(
    derivation = "Unchanged thesis-era dead-woody load carried forward from calculate_fuel_load.rda"
  )

# Phase comes from the identifiers retained above. Restrict the exported long
# component table to reportable component/phase combinations.
fuel_components_transect <- bind_rows(
  forest_floor_components,
  vegetation_components,
  dead_woody_components

  mutate(
    component = factor(
      component,

# Explicit pre/post joins replace the thesis-era lexical phase ordering used to
# construct veg_diff. The primary PCT contrast is woody standing vegetation
# reduction; it is an inferred surface-fuel contribution, not direct slash load.
vegetation_wide <- fuel_components_transect |>
  filter(component %in% c("woody_vegetation", "herbaceous_vegetation")) |>
  select(all_of(ids), phase, component, load_mgha) |>
  pivot_wider(
    names_from = c(component, phase),
    values_from = load_mgha,
    names_sep = "_"
  )

fuel_bed_depth_phase <- forest_floor_station |>
  summarise(
    fuel_bed_depth_cm = mean(fuel_bed_depth_cm),
    n_fuel_bed_stations = sum(!is.na(fuel_bed_depth_cm)),
    .by = c("phase", all_of(ids))
  ) |>
  pivot_wider(
    names_from = phase,
    values_from = c(fuel_bed_depth_cm, n_fuel_bed_stations),
    names_sep = "_"
  )

fuel_paired_transect <- vegetation_wide |>
  filter(
    !is.na(woody_vegetation_prepct),
    !is.na(woody_vegetation_postpct)
  ) |>
  inner_join(fuel_bed_depth_phase, by = ids) |>
  mutate(
    woody_standing_reduction_mgha = woody_vegetation_prepct - woody_vegetation_postpct,
    herbaceous_vegetation_change_mgha = herbaceous_vegetation_postpct - herbaceous_vegetation_prepct,
    interpretation = "Woody standing-vegetation reduction is an inferred potential surface-fuel contribution; it is not a direct slash-load measurement."
  )

pairing_qc <- bind_rows(
  transmute(distinct(filter(vegetation_wide, !is.na(woody_vegetation_prepct)), across(all_of(ids))), status = "pre-PCT woody observation"),
  transmute(distinct(filter(vegetation_wide, !is.na(woody_vegetation_postpct)), across(all_of(ids))), status = "post-PCT woody observation"),

02 — Explore component distributions

Purpose. Describe zeros, spread, and replication before fitting.

Rationale. Keep the design random intercepts fixed. The exploration candidate CSVs are advisory specifications; the fitting stage below explicitly constructs its candidate set from the data.

Calculation — f01-02-explore-components 
local({
publication_root <- normalizePath(getwd(), winslash = "/")

add_design_groups <- function(data) {
  data |>
    mutate(
      block = interaction(site, treatment, drop = TRUE),
      design_corner = interaction(site, treatment, corner, drop = TRUE)
    )
}

component_data <- components |>
  mutate(component = as.character(component)) |>
  add_design_groups()

component_summary <- component_data |>
  summarise(
    n_transects = n(),
    n_sites = n_distinct(site),
    n_blocks = n_distinct(block),
    n_corners = n_distinct(design_corner),
    n_zero = sum(load_mgha == 0, na.rm = TRUE),
    prop_zero = mean(load_mgha == 0, na.rm = TRUE),
    n_missing = sum(is.na(load_mgha)),
    mean_mgha = mean(load_mgha, na.rm = TRUE),
    sd_mgha = sd(load_mgha, na.rm = TRUE),
    variance_mgha = var(load_mgha, na.rm = TRUE),
    cv = if_else(mean_mgha > 0, sd_mgha / mean_mgha, NA_real_),
    min_mgha = min(load_mgha, na.rm = TRUE),
    q25_mgha = quantile(load_mgha, 0.25, na.rm = TRUE, names = FALSE),
    median_mgha = median(load_mgha, na.rm = TRUE),
    q75_mgha = quantile(load_mgha, 0.75, na.rm = TRUE, names = FALSE),
    max_mgha = max(load_mgha, na.rm = TRUE),
    .by = c("phase", "component", "treatment")
  ) |>
  arrange(phase, component, treatment)

component_candidates <- component_data |>
  summarise(
    n_zero = sum(load_mgha == 0, na.rm = TRUE),
    n_transects = n(),
    n_missing = sum(is.na(load_mgha)),
    .by = c("phase", "component")
  ) |>
  mutate(
    response_type = "nonnegative continuous load",
    fixed_effects = "treatment",
    random_effects = "(1 | site) + (1 | site:treatment) + (1 | site:treatment:corner)",
    conditional_family = if_else(n_zero > 0, "ziGamma(link = log)", "Gamma(link = log)"),
    zero_inflation_candidates = if_else(
      n_zero > 0,
      "~1; consider ~treatment only if zero-process diagnostics and replication support it",
      "none unless later diagnostics identify a structural-zero process"
    ),
    dispersion_candidates = "~1 first; compare ~treatment only if residual dispersion diagnostics support it",
    status = if_else(
      n_missing > 0,
      "candidate specifications only; resolve missing values before model fitting",
      "candidate specifications only; no model fitted"
    )
  ) |>
  arrange(phase, component)

fit_component_candidates <- function(data, phase_value, component_value) {
  model_data <- data |>
    filter(
      .data$phase == .env$phase_value,
      .data$component == .env$component_value,
      !is.na(load_mgha)
    )
  zero_count <- sum(model_data$load_mgha == 0)
  prefix <- paste(phase_value, component_value, sep = "_")

  candidates <- list()
  if (zero_count > 0) {
    candidates[["base"]] <- list(glmmTMB::ziGamma(link = "log"), "ziGamma(link = log)", ~1, "~1", ~1, "~1")
    candidates[["disp_treatment"]] <- list(glmmTMB::ziGamma(link = "log"), "ziGamma(link = log)", ~1, "~1", ~treatment, "~treatment")
    if (zero_count >= 8) {
      candidates[["zi_treatment"]] <- list(glmmTMB::ziGamma(link = "log"), "ziGamma(link = log)", ~treatment, "~treatment", ~1, "~1")
    }
  } else {
    candidates[["base"]] <- list(Gamma(link = "log"), "Gamma(link = log)", ~0, "~0", ~1, "~1")
    candidates[["disp_treatment"]] <- list(Gamma(link = "log"), "Gamma(link = log)", ~0, "~0", ~treatment, "~treatment")
  }

  # The thesis used site plus treatment in both the zero and dispersion
  # submodels for post-PCT 100-hour fuels. Retain that candidate and add one
  # deliberately more complex site-by-treatment dispersion sensitivity check.
  if (phase_value == "postpct" && component_value == "hundhr") {

      glmmTMB::ziGamma(link = "log"), "ziGamma(link = log)",
      ~treatment + site, "~treatment + site",
      ~treatment + site, "~treatment + site"
    )
    candidates[["zi_treatment_site_disp_treatment_site_interaction"]] <- list(
      glmmTMB::ziGamma(link = "log"), "ziGamma(link = log)",
      ~treatment + site, "~treatment + site",
      ~treatment * site, "~treatment * site"
    )
  }

  imap_dfr(candidates, function(candidate, candidate_name) {
    fit_and_diagnose(
      simulation_n = 1000, simulation_seed = 20260918,
      model_id = paste(prefix, candidate_name, sep = "_"),

      ziformula = candidate[[3]],
      ziformula_label = candidate[[4]],
      dispformula = candidate[[5]],
      dispformula_label = candidate[[6]],
      paths = paths
    )
  })
}

publication_paths <- function(root) {
  root <- normalizePath(root, winslash = "/", mustWork = TRUE)
  staged <- Sys.getenv("PUBLICATION_OUTPUT_ROOT")
  output <- if (nzchar(staged)) staged else file.path(root, "outputs")
  dir.create(output, recursive = TRUE, showWarnings = FALSE)
  list(root = root, input = file.path(root, "inputs", "thesis"),
       output = normalizePath(output, winslash = "/"))
}

model_output_paths <- function(paths) {
  models <- file.path(paths$output, "models")
  diagnostics <- file.path(paths$output, "diagnostics")
  dir.create(models, recursive = TRUE, showWarnings = FALSE)
  dir.create(diagnostics, recursive = TRUE, showWarnings = FALSE)
  list(models = models, diagnostics = diagnostics)
}

extract_test <- function(test_fun, simulation) {
  result <- tryCatch(test_fun(simulation), error = function(error) error)
  if (inherits(result, "error")) {
    return(list(statistic = NA_real_, p_value = NA_real_, error = conditionMessage(result)))
  }
  list(
    statistic = unname(result$statistic)[1],
    p_value = result$p.value,
    error = NA_character_
  )
}

fit_and_diagnose <- function(
    model_id,
    response,
    analysis,
    data,
    formula,
    family,
    family_label,
    ziformula,
    ziformula_label,
    dispformula,
    dispformula_label,
    paths,
    simulation_n,
    simulation_seed) {
  output_paths <- model_output_paths(paths)
  warnings <- character()
  fit <- tryCatch(
    withCallingHandlers(
      glmmTMB::glmmTMB(
        formula = formula,
        data = data,
        family = family,
        ziformula = ziformula,
        dispformula = dispformula
      ),
      warning = function(warning) {
        warnings <<- c(warnings, conditionMessage(warning))
        invokeRestart("muffleWarning")
      }
    ),
    error = function(error) error
  )

  base_record <- tibble::tibble(
    model_id = model_id,
    response = response,
    analysis = analysis,
    n = nrow(data),
    fixed_and_random_formula = paste(deparse(formula), collapse = " "),
    family = family_label,
    ziformula = ziformula_label,
    dispformula = dispformula_label,
    fit_status = if (inherits(fit, "error")) "error" else "fit",
    fit_message = if (inherits(fit, "error")) conditionMessage(fit) else NA_character_,
    warnings = if (length(warnings)) paste(unique(warnings), collapse = " | ") else NA_character_,
    pd_hessian = NA,
    aic = NA_real_,
    dharma_uniformity_p = NA_real_,
    dharma_dispersion_p = NA_real_,
    dharma_outlier_p = NA_real_,
    dharma_zero_inflation_p = NA_real_,
    diagnostic_message = NA_character_
  )

  if (inherits(fit, "error")) {
    return(base_record)
  }

  saveRDS(fit, file.path(output_paths$models, paste0(model_id, ".rds")))
  simulation <- tryCatch(
    publication_simulate_residuals(fit, n = simulation_n, seed = simulation_seed),
    error = function(error) error
  )
  if (inherits(simulation, "error")) {
    return(dplyr::mutate(
      base_record,
      pd_hessian = fit$sdr$pdHess,
      aic = stats::AIC(fit),
      diagnostic_message = conditionMessage(simulation)
    ))
  }

  # Some DHARMa tests may draw internally. Use a null device so these
  # exploratory diagnostics never create an incidental Rplots.pdf in analysis/.
  grDevices::pdf(file = NULL)
  on.exit(grDevices::dev.off(), add = TRUE)
  uniformity <- extract_test(DHARMa::testUniformity, simulation)
  dispersion <- extract_test(DHARMa::testDispersion, simulation)
  outliers <- extract_test(DHARMa::testOutliers, simulation)
  zero_inflation <- extract_test(DHARMa::testZeroInflation, simulation)

  png(
    filename = file.path(output_paths$diagnostics, paste0(model_id, "-dharma.png")),
    width = 1800,
    height = 900,
    res = 150
  )
  plot(simulation)
  dev.off()

  dplyr::mutate(
    base_record,
    pd_hessian = fit$sdr$pdHess,
    aic = stats::AIC(fit),
    dharma_uniformity_p = uniformity$p_value,
    dharma_dispersion_p = dispersion$p_value,
    dharma_outlier_p = outliers$p_value,
    dharma_zero_inflation_p = zero_inflation$p_value,
    diagnostic_message = paste(
      stats::na.omit(c(uniformity$error, dispersion$error, outliers$error, zero_inflation$error)),
      collapse = " | "
    )
  )
}

# Shared runtime check: inspect actual namespaces and readable DESCRIPTION files.
verify_publication_environment <- function() {
  root <- normalizePath(getwd(), winslash = "/")
  stopifnot(getRversion() == "4.6.2", file.exists("renv.lock"),
            grepl(paste0(root, "/renv/library/"),
                  normalizePath(.libPaths()[1], winslash = "/"), fixed = TRUE),
            !any(grepl("analysis/renv", .libPaths(), fixed = TRUE)))
  packages <- trimws(strsplit(gsub("[\r\n]", "", read.dcf("DESCRIPTION")[1, "Imports"]), ",")[[1]])
  missing <- packages[!vapply(packages, requireNamespace, logical(1), quietly = TRUE)]
  if (length(missing)) stop("Required namespaces unavailable: ", paste(missing, collapse = ", "))
  lock <- jsonlite::fromJSON("renv.lock", simplifyVector = FALSE)
  for (package in names(lock$Packages)) {
    descriptions <- file.path(.libPaths(), package, "DESCRIPTION")
    descriptions <- descriptions[file.exists(descriptions)]
    version <- if (length(descriptions)) unname(read.dcf(descriptions[1], fields = "Version")[1, 1]) else NA_character_
    if (!identical(version, lock$Packages[[package]]$Version)) stop("Package differs from publication lock: ", package)
  }
  invisible(TRUE)
}
